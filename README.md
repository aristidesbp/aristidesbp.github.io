# Criar banco de dados (usar editor)

```
-- CRIE AS TABELA categoria:
CREATE TABLE public.categoria (
id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
categoria text,
created_at timestamp with time zone NOT NULL DEFAULT now(),
CONSTRAINT categoria_pkey PRIMARY KEY (id)
);

-- CRIE AS TABELA usuarios:
CREATE TABLE public.usuarios (
id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,user_id uuid NOT NULL UNIQUE,
nome_completo text,
unidade text,
bio text,
avatar_url text,
created_at timestamp with time zone DEFAULT now(),
CONSTRAINT usuarios_pkey PRIMARY KEY (id),
CONSTRAINT profiles_user_id_fkey FOREIGN KEY (user_id) REFERENCES
auth.users(id)
);

-- CRIE AS TABELA servicos:
CREATE TABLE public.servicos (
id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
criado_por bigint NOT NULL,
titulo text NOT NULL,
descricao text NOT NULL,
categoria bigint NOT NULL,
preco_estimado text,
foto_url text,
created_at timestamp with time zone DEFAULT now(),
terceiro boolean DEFAULT true,
eu_mesmo boolean DEFAULT false,
whatsapp text,
preco_detalhe text,
CONSTRAINT servicos_pkey PRIMARY KEY (id),
CONSTRAINT servicos_categoria_fkey FOREIGN KEY (categoria) REFERENCES
public.categoria(id),
CONSTRAINT servicos_criado_por_fkey FOREIGN KEY (criado_por) REFERENCES
public.usuarios(id)
);

-- CRIE AS TABELA avaliacoes:
CREATE TABLE public.avaliacoes (
id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
servico_id bigint NOT NULL,
autor_id bigint NOT NULL,
nota integer CHECK (nota >= 1 AND nota <= 5),
comentario text,
created_at timestamp with time zone DEFAULT now(),
CONSTRAINT avaliacoes_pkey PRIMARY KEY (id),
CONSTRAINT avaliacoes_servico_id_fkey FOREIGN KEY (servico_id)
REFERENCES public.servicos(id),
CONSTRAINT avaliacoes_autor_id_fkey FOREIGN KEY (autor_id) REFERENCES
public.usuarios(id)
);

-- CRIE AS TABELA favoritos:
CREATE TABLE public.favoritos (
id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
usuario_id bigint NOT NULL,
servico_id bigint NOT NULL,
created_at timestamp with time zone DEFAULT now(),
CONSTRAINT favoritos_pkey PRIMARY KEY (id),
CONSTRAINT favoritos_usuario_id_fkey FOREIGN KEY (usuario_id)
REFERENCES public.usuarios(id),
CONSTRAINT favoritos_servico_id_fkey FOREIGN KEY (servico_id)
REFERENCES public.servicos(id)
);
```

```
CREATE TABLE IF NOT EXISTS produtos (

    fornecedor_id uuid REFERENCES entidades(id), -- Ligação à tabela entidades
    especificacoes jsonb,

    unidade_comercial character varying,

-- 4. Tabela: Auditoria de Produtos (Histórico de alterações)[cite: 2]
CREATE TABLE IF NOT EXISTS auditoria_produtos (

-- 5. Tabela: Terminais do PDV[cite: 2]
CREATE TABLE IF NOT EXISTS pdv_terminais (

-- =========================================================================
-- 🗂️ PARTE 2: CRIAÇÃO DO STORAGE (BUCKETS)[cite: 2]
-- =========================================================================
-- Nota: O Supabase gere os buckets numa tabela interna chamada storage.buckets.

INSERT INTO storage.buckets (id, name, public) 
VALUES 
  ('foto_entidades', 'foto_entidades', true),
  ('foto_produtos', 'foto_produtos', true),
  ('outras_midias', 'outras_midias', true)
ON CONFLICT (id) DO NOTHING; -- Se já existir, ignora para não dar erro.


-- =========================================================================
-- 🛡️ PARTE 3: SEGURANÇA (RLS - Row Level Security)
-- =========================================================================

-- Ativar o RLS nas tabelas essenciais para que as regras funcionem
ALTER TABLE entidades ENABLE ROW LEVEL SECURITY;
ALTER TABLE produtos ENABLE ROW LEVEL SECURITY;
ALTER TABLE pdv_vendas ENABLE ROW LEVEL SECURITY;
ALTER TABLE fin_contas ENABLE ROW LEVEL SECURITY;

-- Regras para 'entidades'[cite: 2]
CREATE POLICY "Autenticados_Gerenciam_Entidades" ON entidades
FOR ALL USING (auth.role() = 'authenticated');

-- Regras para 'produtos'[cite: 2]
CREATE POLICY "Leitura_Publica_Vitrine" ON produtos
FOR SELECT USING ((esta_ativo = true) AND (visivel_vitrine = true));

CREATE POLICY "Autenticados_Gerenciam_Produtos" ON produtos
FOR ALL USING (auth.role() = 'authenticated');

-- Regras para 'pdv_vendas'[cite: 2]
CREATE POLICY "Anon_Insere_Vendas" ON pdv_vendas
FOR INSERT WITH CHECK (true); -- Permite inserir sem estar autenticado (útil para vitrine pública)

CREATE POLICY "Autenticados_Gere_Vendas" ON pdv_vendas
FOR ALL USING (auth.role() = 'authenticated');

-- Regras para 'fin_contas'[cite: 2]
CREATE POLICY "Autenticados_Gerenciam_Financas" ON fin_contas
FOR ALL USING (auth.role() = 'authenticated');


-- =========================================================================
-- 🖼️ PARTE 4: POLÍTICAS DE SEGURANÇA DO STORAGE[cite: 2]
-- =========================================================================

-- Permitir leitura pública das imagens
CREATE POLICY "Leitura_Publica_Fotos" ON storage.objects
FOR SELECT USING (bucket_id IN ('foto_entidades', 'foto_produtos', 'outras_midias'));

-- Permitir inserção (Upload) para utilizadores autenticados
CREATE POLICY "Autenticados_Sobem_Fotos" ON storage.objects
FOR INSERT WITH CHECK (auth.role() = 'authenticated');

-- Permitir atualização de ficheiros para utilizadores autenticados
CREATE POLICY "Autenticados_Atualizam_Fotos" ON storage.objects
```

```
-- [INÍCIO: EXTRATOR_DE_SCHEMA_SUPABASE]
-- Este script consulta os metadados do PostgreSQL para criar um raio-x do seu schema 'public'

WITH tables_info AS (
    -- Busca todas as tabelas públicas e junta as suas colunas em uma lista
    SELECT jsonb_agg(jsonb_build_object(
        'tabela', t.table_name,
        'colunas', (
            SELECT jsonb_agg(c.column_name || ' (' || c.data_type || ')')
            FROM information_schema.columns c
            WHERE c.table_name = t.table_name AND c.table_schema = 'public'
        )
    )) AS dados
    FROM information_schema.tables t
    WHERE t.table_schema = 'public'
),
rls_info AS (
    -- Busca todas as regras de segurança RLS (Row Level Security)
    SELECT COALESCE(jsonb_agg(jsonb_build_object(
        'tabela', tablename,
        'politica', policyname,
        'comando', cmd,
        'roles', roles,
        'condicao', qual
    )), '[]'::jsonb) AS dados
    FROM pg_policies
    WHERE schemaname = 'public'
),
rpc_info AS (
    -- Busca todas as Funções (RPCs) criadas no schema público
    SELECT COALESCE(jsonb_agg(jsonb_build_object(
        'funcao', p.proname,
        'argumentos', pg_get_function_arguments(p.oid),
        'retorno', pg_get_function_result(p.oid)
    )), '[]'::jsonb) AS dados
    FROM pg_proc p
    JOIN pg_namespace n ON p.pronamespace = n.oid
    WHERE n.nspname = 'public'
),
triggers_info AS (
    -- Busca todos os Triggers (gatilhos) atrelados às tabelas
    SELECT COALESCE(jsonb_agg(jsonb_build_object(
        'tabela', event_object_table,
        'trigger', trigger_name,
        'evento', event_manipulation,
        'tempo', action_timing
    )), '[]'::jsonb) AS dados
    FROM information_schema.triggers
    WHERE trigger_schema = 'public'
)
-- Compila todas as informações acima em um único objeto JSON formatado
SELECT jsonb_pretty(jsonb_build_object(
    '1_tabelas', (SELECT dados FROM tables_info),
    '2_rls_politicas', (SELECT dados FROM rls_info),
    '3_funcoes_rpc', (SELECT dados FROM rpc_info),
    '4_gatilhos_triggers', (SELECT dados FROM triggers_info)
)) AS relatorio_completo;

-- [FIM: EXTRATOR_DE_SCHEMA_SUPABASE]
```

```
 * Limpa textos vindos do utilizador ou do banco para impedir ataques de código malicioso (XSS).
 */
function escaparHTML(texto) {
    if (texto === null || texto === undefined) return '';
    return String(texto)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
}

/**
 * Mostra uma notificação flutuante bonita no canto superior direito.
 * @param {string} mensagem - O texto a exibir.
 * @param {string} tipo - 'sucesso', 'erro', 'info' ou 'aviso'.
 */
function mostrarToast(mensagem, tipo = 'sucesso') {
    const container = document.getElementById('toast-container');
    if (!container) return;

    const toast = document.createElement('div');
    toast.className = `toast toast-${tipo}`;
    
    let icone = '✅';
    if (tipo === 'erro') icone = '❌';
    if (tipo === 'info') icone = 'ℹ️';
    if (tipo === 'aviso') icone = '⚠️';

    toast.innerHTML = `<span>${icone}</span> <span>${escaparHTML(mensagem)}</span>`;
    container.appendChild(toast);

    // Remove o aviso automaticamente após 4 segundos
    setTimeout(() => {
        toast.style.opacity = '0';
        toast.style.transform = 'translateX(120%)';
        toast.style.transition = 'all 0.3s ease';
        setTimeout(() => toast.remove(), 300);
    }, 4000);
}

/**
 * Traduz erros técnicos do banco de dados para mensagens claras em português.
 */
function tratarErroSupabase(erro, contexto = 'Operação') {
    console.error(`[Erro em ${contexto}]:`, erro);
    const msg = (erro && erro.message) ? erro.message.toLowerCase() : String(erro).toLowerCase();

    if (msg.includes('foreign key constraint')) {
        mostrarToast('Não é possível apagar este registo porque ele está vinculado a vendas, produtos ou lançamentos financeiros. Use a opção de enviar para a Lixeira!', 'aviso');
    } else if (msg.includes('duplicate key') || msg.includes('unique constraint')) {
        mostrarToast('Já existe um registo cadastrado com este Código, Documento ou E-mail.', 'aviso');
    } else if (msg.includes('invalid login credentials')) {
        mostrarToast('E-mail ou palavra-passe incorretos. Tente novamente.', 'erro');
    } else if (msg.includes('jwt expired') || msg.includes('not authenticated')) {
        mostrarToast('A sua sessão expirou por segurança. Faça login novamente.', 'aviso');
        iniciarApp();
    } else {
        mostrarToast(`${contexto}: Ocorreu uma falha de comunicação (${erro.message || 'Erro desconhecido'}).`, 'erro');
    }
}

/* --- 4. COMPRESSOR DE IMAGENS WEBP E NOMES SEGUROS --- */
function comprimirParaWebP(file, maxWidthHeight = 800) {
    return new Promise((resolve, reject) => {
        if (!file) return reject(new Error("Nenhum ficheiro fornecido."));
        const reader = new FileReader();
        reader.readAsDataURL(file);
        reader.onload = (event) => {
            const img = new Image();
            img.src = event.target.result;
            img.onload = () => {
                let width = img.width;
                let height = img.height;
                if (width > height) {
                    if (width > maxWidthHeight) { height = Math.round(height * (maxWidthHeight / width)); width = maxWidthHeight; }
                } else {
                    if (height > maxWidthHeight) { width = Math.round(width * (maxWidthHeight / height)); height = maxWidthHeight; }
                }
                const canvas = document.createElement('canvas');
                canvas.width = width;
                canvas.height = height;
                const ctx = canvas.getContext('2d');
                ctx.drawImage(img, 0, 0, width, height);
                canvas.toBlob((blob) => {
                    if (blob) {
                        const novoNome = file.name.replace(/\.[^/.]+$/, ".webp");
                        resolve(new File([blob], novoNome, { type: 'image/webp' }));
                    } else {
                        reject(new Error("Falha ao converter imagem para WebP."));
                    }
                }, 'image/webp', 0.8);
            };
            img.onerror = () => reject(new Error("Formato de imagem inválido."));
        };
        reader.onerror = () => reject(new Error("Erro ao ler ficheiro."));
    });
}

function gerarNomeSeguroArquivo(prefixo) {
    let nomeLimpo = (prefixo || 'arq').normalize('NFD').replace(/[\u0300-\u036f]/g, "").toLowerCase();
    nomeLimpo = nomeLimpo.replace(/[^a-z0-9]/g, '-').replace(/-+/g, '-');
    return `${nomeLimpo}-${Date.now()}.webp`;
}

function abrirScannerGlobal(callback) {
    scannerCallbackDestino = callback;
    const modal = document.getElementById('modal-scanner-global');
    modal.style.display = 'flex';

    if (!scannerGlobalInstancia) {
        scannerGlobalInstancia = new Html5Qrcode("leitor-scanner-global");
    }

    scannerGlobalInstancia.start(
        { facingMode: "environment" },
        { fps: 10, qrbox: { width: 250, height: 110 } },
        (codigoLido) => {
            fecharScannerGlobal();
            mostrarToast(`Código lido: ${codigoLido}`, 'info');
            if (scannerCallbackDestino) scannerCallbackDestino(codigoLido);
        },
        () => {}
    ).catch((err) => {
        mostrarToast("Não foi possível aceder à câmara. Verifique as permissões do navegador.", "erro");
        fecharScannerGlobal();
    });
}

function fecharScannerGlobal() {
    const modal = document.getElementById('modal-scanner-global');
    if (scannerGlobalInstancia) {
        scannerGlobalInstancia.stop().then(() => { modal.style.display = 'none'; }).catch(() => { modal.style.display = 'none'; });
    } else {
        modal.style.display = 'none';
    }
}

/* --- 6. CARREGAMENTO AUTOMÁTICO DOS DADOS DA EMPRESA --- */
async function carregarConfiguracoesEmpresa() {
    try {
        // Carrega preferências locais complementares (Margem XML e PIX)
        const configLocal = localStorage.getItem('erp_config_complementar');
        if (configLocal) {
            const parsed = JSON.parse(configLocal);
            dadosEmpresaGlobal.margem_padrao_xml = parseFloat(parsed.margem_padrao_xml) || 30;
            dadosEmpresaGlobal.chave_pix = parsed.chave_pix || '';
        }

        // Busca os dados oficiais da empresa na tabela 'entidades' (tipo = 'empresa')
        const { data, error } = await clienteSupabase
            .from('entidades')
            .select('*')
            .eq('tipo', 'empresa')
            .neq('status', 'lixeira')
            .limit(1)
            .maybeSingle();

        if (data && !error) {
            dadosEmpresaGlobal.id = data.id;
            dadosEmpresaGlobal.nome_completo = data.nome_completo || 'Minha Loja ERP';
            dadosEmpresaGlobal.numero_documento = data.numero_documento || '';
            dadosEmpresaGlobal.telefone = data.telefone || '5591999999999';
            dadosEmpresaGlobal.email = data.email || '';
            dadosEmpresaGlobal.endereco_completo = data.endereco_completo || '';
            dadosEmpresaGlobal.foto_url = data.foto_url || '';
        }
    } catch (e) {
        console.log("Aviso: Dados da empresa ainda não configurados.");
    }
}
```

```
async function carregarListaEntidadesAtivas() {
    const container = document.getElementById('ent-lista-container');
    if (!container) return;

    const termo = document.getElementById('ent-busca').value.trim();
    const tipo = document.getElementById('ent-filtro-tipo').value;
    const limite = parseInt(document.getElementById('ent-itens-por-pagina').value) || 15;

    container.innerHTML = "<p style='text-align:center; color: var(--text-muted);'>⏳ A consultar Supabase...</p>";
    
    const inicio = paginaAtualEnt * limite;
    const fim = inicio + limite - 1;

    try {
        let query = clienteSupabase
            .from('entidades')
            .select('*', { count: 'exact' })
            .neq('status', 'lixeira')
            .order('nome_completo', { ascending: true })
            .range(inicio, fim);

        if (tipo) query = query.eq('tipo', tipo);
        if (termo) query = query.or(`nome_completo.ilike.%${termo}%,numero_documento.ilike.%${termo}%`);

        const { data, count, error } = await query;
        if (error) throw error;

        atualizarPaginacaoEntidades(count, limite);
        container.innerHTML = "";

        if (!data || data.length === 0) {
            container.innerHTML = "<p style='text-align:center; color: var(--text-muted);'>Nenhuma entidade encontrada.</p>";
            return;
        }

        data.forEach(ent => renderizarCartaoEntidade(ent, container, false));
    } catch (erro) {
        tratarErroSupabase(erro, "Listar Entidades");
        container.innerHTML = "<p style='text-align:center; color: var(--danger-color);'>❌ Erro ao carregar dados.</p>";
    }
}

async function esvaziarLixeiraEntidadesDefinitivo() {

        const { data: itensLixeira } = await clienteSupabase.from('entidades').select('foto_url').eq('status', 'lixeira');

        if (itensLixeira && itensLixeira.length > 0) {

                .filter(i => i.foto_url)
                .map(i => decodeURIComponent(i.foto_url.split('/').pop().split('?')[0]));

                await clienteSupabase.storage.from('foto_entidades').remove(fotosParaRemover);

        carregarListaEntidadesLixeira();

        tratarErroSupabase(e, "Purgar Lixeira");
```

```
async function carregarListaProdutosAtivos() {
    const container = document.getElementById('prod-lista-container');
    if (!container) return;

    const termo = document.getElementById('prod-busca').value.trim();
    const limite = parseInt(document.getElementById('prod-itens-por-pagina').value) || 15;

    container.innerHTML = "<p style='text-align:center; color: var(--text-muted);'>⏳ A consultar Supabase...</p>";
    const inicio = paginaAtualProd * limite;
    const fim = inicio + limite - 1;

    try {
        let query = clienteSupabase
            .from('produtos')
            .select('*', { count: 'exact' })
            .eq('esta_ativo', true)
            .order('nome', { ascending: true })
            .range(inicio, fim);

        if (termo) {
            query = query.or(`nome.ilike.%${termo}%,codigo_barras.ilike.%${termo}%,sku.ilike.%${termo}%`);
        }

        const { data, count, error } = await query;
        if (error) throw error;

        atualizarPaginacaoProdutos(count, limite);
        container.innerHTML = "";

        if (!data || data.length === 0) {
            container.innerHTML = "<p style='text-align:center; color: var(--text-muted);'>Nenhum produto encontrado no catálogo.</p>";
            return;
        }

        data.forEach(prod => renderizarCartaoProduto(prod, container, false));
    } catch (e) {
        tratarErroSupabase(e, "Listar Produtos");
        container.innerHTML = "<p style='text-align:center; color: var(--danger-color);'>❌ Erro ao carregar produtos.</p>";
    }
}

async function esvaziarLixeiraProdutosDefinitivo() {

        carregarListaProdutosLixeira();

function desenharModuloImportarXml() {
    appRoot.innerHTML = `

                <p><strong>CNPJ:</strong> <span id="xml-resumo-cnpj">-</span></p>
                <p><strong>Chave da Nota:</strong> <span id="xml-resumo-chave" style="font-size: 0.85em; word-break: break-all;">-</span></p>
                <p><strong>Itens Encontrados:</strong> <span id="xml-resumo-qtd">0</span> produtos</p>
                
                <button id="btn-processar-xml" class="btn-sucesso" style="margin-top: 15px;">🚀 Confirmar e Importar Produtos para o Stock</button>
            </div>

            <button type="button" class="btn-secundario" onclick="desenharModuloProdutos()">⬅️ Voltar ao Catálogo</button>
        </div>
    `;

    let dadosXmlProcessados = null;

    document.getElementById('input-arquivo-xml').addEventListener('change', (e) => {
        const file = e.target.files[0];
        if (!file) return;

        document.getElementById('xml-nome-arquivo').textContent = file.name;
        const reader = new FileReader();
        reader.onload = (event) => {
            try {
                const parser = new DOMParser();
                const xmlDoc = parser.parseFromString(event.target.result, "text/xml");

                // 1. Ler dados do Emitente (Fornecedor)
                const emit = xmlDoc.getElementsByTagName("emit")[0] || xmlDoc.getElementsByTagName("TNFE.emit")[0];
                if (!emit) throw new Error("Estrutura de emitente não encontrada no XML.");

                const nomeForn = emit.getElementsByTagName("xNome")[0]?.textContent || "Fornecedor Desconhecido";
                const cnpjForn = emit.getElementsByTagName("CNPJ")[0]?.textContent || "";

                // 2. Ler Chave da NFe
                const infNFe = xmlDoc.getElementsByTagName("infNFe")[0];
                const chaveNFe = infNFe ? infNFe.getAttribute("Id") : "";

                // 3. Ler Itens (Produtos)
                const dets = xmlDoc.getElementsByTagName("det");
                if (dets.length === 0) throw new Error("Nenhum item detetado na nota fiscal.");

                let itensArray = [];
                for (let i = 0; i < dets.length; i++) {
                    const prod = dets[i].getElementsByTagName("prod")[0];
                    const cProd = prod.getElementsByTagName("cProd")[0]?.textContent || '';
                    const cEAN = prod.getElementsByTagName("cEAN")[0]?.textContent || '';
                    const xProd = prod.getElementsByTagName("xProd")[0]?.textContent || 'Produto sem nome';
                    const NCM = prod.getElementsByTagName("NCM")[0]?.textContent || '';
                    const CFOP = prod.getElementsByTagName("CFOP")[0]?.textContent || '';
                    const uCom = prod.getElementsByTagName("uCom")[0]?.textContent || 'UN';
                    const qCom = parseFloat(prod.getElementsByTagName("qCom")[0]?.textContent || 1);
                    const vUnCom = parseFloat(prod.getElementsByTagName("vUnCom")[0]?.textContent || 0);

                    // Calcula preço de venda aplicando a margem cadastrada nas configurações da empresa
                    const margem = (dadosEmpresaGlobal.margem_padrao_xml || 30) / 100;
                    const precoVendaSugerido = vUnCom * (1 + margem);

                    itensArray.push({
                        sku: cProd,
                        ean: (cEAN && cEAN !== 'SEM GTIN') ? cEAN : null,
                        nome: xProd,
                        ncm: NCM,
                        cfop: CFOP,
                        unidade: uCom,
                        quantidade: qCom,
                        preco_custo: vUnCom,
                        preco_venda: parseFloat(precoVendaSugerido.toFixed(2))
                    });
                }

                dadosXmlProcessados = {
                    fornecedor: { nome: nomeForn, cnpj: cnpjForn },
                    chave: chaveNFe,
                    itens: itensArray
                };

                document.getElementById('xml-resumo-forn').textContent = nomeForn;
                document.getElementById('xml-resumo-cnpj').textContent = cnpjForn;
                document.getElementById('xml-resumo-chave').textContent = chaveNFe;
                document.getElementById('xml-resumo-qtd').textContent = itensArray.length;
                document.getElementById('xml-preview-container').style.display = 'block';

                mostrarToast("XML lido com sucesso! Verifique o resumo.", "sucesso");
            } catch (err) {
                tratarErroSupabase(err, "Leitura do XML");
            }
        };
        reader.readAsText(file);
    });

    document.getElementById('btn-processar-xml')?.addEventListener('click', async () => {
        if (!dadosXmlProcessados) return;
        const btn = document.getElementById('btn-processar-xml');
        btn.textContent = "⏳ A importar para o stock...";
        btn.disabled = true;

        try {
            // 1. Verifica ou cadastra o fornecedor automaticamente
            let fornecedorId = null;
            if (dadosXmlProcessados.fornecedor.cnpj) {
                const { data: fornExistente } = await clienteSupabase
                    .from('entidades')
                    .select('id')
                    .eq('numero_documento', dadosXmlProcessados.fornecedor.cnpj)
                    .maybeSingle();

                if (fornExistente) {
                    fornecedorId = fornExistente.id;
                } else {
                    const { data: novoForn, error: errForn } = await clienteSupabase
                        .from('entidades')
                        .insert([{
                            tipo: 'fornecedor',
                            nome_completo: dadosXmlProcessados.fornecedor.nome,
                            tipo_documento: 'CNPJ',
                            numero_documento: dadosXmlProcessados.fornecedor.cnpj,
                            status: 'ativo'
                        }])
                        .select('id')
                        .single();
                    if (!errForn && novoForn) fornecedorId = novoForn.id;
                }
            }

            // 2. Processa cada item do XML usando a função RPC existente no Supabase
            let itensImportados = 0;
            for (const item of dadosXmlProcessados.itens) {
                // Tenta encontrar produto existente por EAN ou SKU
                let produtoExistente = null;
                if (item.ean) {
                    const { data } = await clienteSupabase.from('produtos').select('*').eq('codigo_barras', item.ean).maybeSingle();
                    produtoExistente = data;
                }
                if (!produtoExistente && item.sku) {
                    const { data } = await clienteSupabase.from('produtos').select('*').eq('sku', item.sku).maybeSingle();
                    produtoExistente = data;
                }

                if (produtoExistente) {
                    // Produto existe: atualiza stock somando a quantidade e atualiza custos
                    const novoEstoque = parseFloat(produtoExistente.estoque_atual || 0) + item.quantidade;
                    await clienteSupabase.rpc('atualizar_produto_seguro', {
                        p_id: produtoExistente.id,
                        p_nome: produtoExistente.nome,
                        p_preco: produtoExistente.preco_venda,
                        p_codigo_produto: produtoExistente.sku,
                        p_ean: produtoExistente.codigo_barras,
                        p_ncm: produtoExistente.ncm,
                        p_unidade_comercial: produtoExistente.unidade_comercial,
                        p_quantidade: novoEstoque,
                        p_preco_custo: item.preco_custo,
                        p_fornecedor: fornecedorId
                    });
                } else {
                    // Produto novo: insere utilizando a função RPC segura
                    await clienteSupabase.rpc('adicionar_produto_seguro', {
                        p_nome: item.nome,
                        p_preco: item.preco_venda,
                        p_codigo_produto: item.sku,
                        p_ean: item.ean,
                        p_ncm: item.ncm,
                        p_unidade_comercial: item.unidade,
                        p_quantidade: item.quantidade,
                        p_preco_custo: item.preco_custo,
                        p_fornecedor: fornecedorId
                    });
                }
                itensImportados++;
            }

            mostrarToast(`Importação concluída! ${itensImportados} produtos processados com sucesso.`, "sucesso");
            desenharModuloProdutos();
        } catch (erro) {
            tratarErroSupabase(erro, "Processar XML");
            btn.textContent = "🚀 Confirmar e Importar Produtos para o Stock";
            btn.disabled = false;
        }
    });
}
```

```
    const inicio = paginaAtualFin * ITENS_PAG_FIN;
    const fim = inicio + ITENS_PAG_FIN - 1;

    try {
        let query = clienteSupabase
            .from('fin_contas')
            .select('*, entidades(nome_completo)', { count: 'exact' })
            .order('data_vencimento', { ascending: true })
            .range(inicio, fim);

        if (tipo) query = query.eq('tipo_conta', tipo);
        if (status) query = query.eq('status_conta', status);
        if (termo) query = query.ilike('descricao', `%${termo}%`);

        const { data, count, error } = await query;
        if (error) throw error;

        const paginas = Math.ceil(count / ITENS_PAG_FIN) || 1;
        document.getElementById('texto-fin-paginacao').textContent = `Página ${paginaAtualFin + 1} de ${paginas}`;

            document.getElementById('fin-valor').value = data.valor_nominal || '';
            document.getElementById('fin-parcelas').value = data.total_parcelas || 1;

            const tipo = document.getElementById('fin-tipo-conta').value;

                    status_conta: status,
                    descricao: desc,
                    entidade_id: entidadeId,
                    valor_nominal: valorTotal,
                    data_vencimento: dataBase
                }).eq('id', idEdicao);
                if (error) throw error;
            } else {
                let arrayLancamentos = [];
                const valorParcela = valorTotal / qtdParcelas;
                for (let i = 1; i <= qtdParcelas; i++) {
                    let d = new Date(dataBase + 'T12:00:00');
                    d.setMonth(d.getMonth() + (i - 1));
                    arrayLancamentos.push({
                        tipo_conta: tipo,
                        status_conta: status,
                        descricao: qtdParcelas > 1 ? `${desc} (${i}/${qtdParcelas})` : desc,
                        entidade_id: entidadeId,
                        valor_nominal: valorParcela,
                        data_emissao: new Date().toISOString().split('T')[0],
                        data_vencimento: d.toISOString().split('T')[0],
                        numero_parcela: i,
                        total_parcelas: qtdParcelas
                    });
                }
                const { error } = await clienteSupabase.from('fin_contas').insert(arrayLancamentos);
                if (error) throw error;
            }

            mostrarToast("Lançamento financeiro gravado com sucesso!", "sucesso");
            desenharModuloFinanceiro();
        } catch (erro) {
            tratarErroSupabase(erro, "Salvar Financeiro");
            btn.textContent = "💾 Salvar Lançamento";
            btn.disabled = false;
        }
    });
}

let turnoAtivoGlobal = null;

async function desenharModuloPDV() {
    appRoot.innerHTML = `
        <div class="card" style="margin-top: 4vh; max-width: 600px; text-align: center;">
            <div class="titulo">🛒 Verificando Caixa (Turno)</div>
            <p style="color: var(--text-muted);">A validar abertura de caixa com o Supabase...</p>
        </div>
    `;

    try {
        const { data: { user } } = await clienteSupabase.auth.getUser();
        // Verifica se existe turno aberto para o operador
        const { data: turno, error } = await clienteSupabase
            .from('pdv_turnos')
            .select('*')
            .eq('operador_id', user.id)
            .eq('status', 'aberto')
            .maybeSingle();

        if (turno) {
            turnoAtivoGlobal = turno;
            desenharTelaCaixaOperacao();
        } else {
            desenharTelaAberturaCaixa(user.id);

        tratarErroSupabase(e, "Verificar Turno PDV");
    }
}

                .insert([{
                    terminal_id: terminalId,
                    operador_id: userId,
                    saldo_abertura: saldo,
                    status: 'aberto'
                }])
                .select('*')
                .single();

async function adicionarProdutoCarrinhoPdv(termo) {
    try {
        const { data, error } = await clienteSupabase
            .from('produtos')
            .select('*')
            .eq('esta_ativo', true)
            .or(`codigo_barras.eq.${termo},sku.eq.${termo},nome.ilike.%${termo}%`)
            .limit(1)
            .maybeSingle();

        if (error || !data) {
            mostrarToast("Produto não encontrado no catálogo.", "aviso");
            return;
        }

        const existente = carrinhoPdv.find(p => p.id === data.id);
        if (existente) {
            existente.quantidade += 1;
            existente.subtotal = existente.quantidade * existente.preco_venda;
        } else {
            carrinhoPdv.push({
                id: data.id,
                nome: data.nome,
                preco_venda: parseFloat(data.preco_venda || 0),
                quantidade: 1,
                subtotal: parseFloat(data.preco_venda || 0),
                estoque_atual: parseFloat(data.estoque_atual || 0)
            });
        }

        renderizarCarrinhoPdv();
        mostrarToast(`${data.nome} adicionado.`, "sucesso");
    } catch (e) {
        tratarErroSupabase(e, "Adicionar ao Carrinho");
    }
}

function renderizarCarrinhoPdv() {
    const container = document.getElementById('pdv-carrinho-itens');
    if (!container) return;

    if (carrinhoPdv.length === 0) {
        container.innerHTML = `<p style='text-align:center; color: var(--text-muted); margin-top: 40px;'>Carrinho vazio. Bipe um produto ou pesquise acima.</p>`;
        document.getElementById('pdv-total-geral').textContent = "R$ 0,00";
        return;
    }

    let html = '';
    let totalGeral = 0;

    carrinhoPdv.forEach((item, index) => {
        totalGeral += item.subtotal;
        html += `
            <div style="background: var(--input-bg); border: 1px solid var(--border-color); padding: 8px 12px; border-radius: 8px; margin-bottom: 8px; display:flex; justify-content:space-between; align-items:center;">
                <div>
                    <strong style="font-size: 0.95em; color: var(--text-main);">${escaparHTML(item.nome)}</strong><br>
                    <span style="font-size: 0.82em; color: var(--text-muted);">${item.quantidade}x R$ ${item.preco_venda.toFixed(2)} = <strong>R$ ${item.subtotal.toFixed(2)}</strong></span>
                </div>
                <div style="display:flex; gap:5px;">
                    <button onclick="alterarQtdCarrinho(${index}, 1)" style="background:var(--info-color); color:#fff; border:none; padding:4px 8px; border-radius:4px; cursor:pointer;">➕</button>
                    <button onclick="alterarQtdCarrinho(${index}, -1)" style="background:var(--warning-color); color:#000; border:none; padding:4px 8px; border-radius:4px; cursor:pointer;">➖</button>
                </div>
            </div>
        `;
    });

    container.innerHTML = html;
    document.getElementById('pdv-total-geral').textContent = `R$ ${totalGeral.toFixed(2)}`;
    calcularTrocoPdv();
}

window.alterarQtdCarrinho = function(index, delta) {
    carrinhoPdv[index].quantidade += delta;
    if (carrinhoPdv[index].quantidade <= 0) {
        carrinhoPdv.splice(index, 1);
    } else {
        carrinhoPdv[index].subtotal = carrinhoPdv[index].quantidade * carrinhoPdv[index].preco_venda;
    }
    renderizarCarrinhoPdv();
};

function calcularTrocoPdv() {
    const total = carrinhoPdv.reduce((acc, item) => acc + item.subtotal, 0);
    const recebido = parseFloat(document.getElementById('pdv-valor-recebido')?.value) || 0;
    const troco = recebido - total;
    const elTroco = document.getElementById('pdv-troco-calc');
    if (elTroco) {
        elTroco.textContent = troco >= 0 ? `R$ ${troco.toFixed(2)}` : "R$ 0,00 (Falta Dinheiro)";
    }
}

async function finalizarVendaPdv() {
    if (carrinhoPdv.length === 0) return mostrarToast("O carrinho está vazio.", "aviso");

    const totalGeral = carrinhoPdv.reduce((acc, item) => acc + item.subtotal, 0);
    const formaPgto = document.getElementById('pdv-forma-pgto').value;
    const valorRecebido = parseFloat(document.getElementById('pdv-valor-recebido').value) || totalGeral;

    if (valorRecebido < totalGeral) {
        return mostrarToast("O valor recebido é inferior ao total da compra.", "aviso");
    }

    const btn = document.getElementById('btn-finalizar-venda');
    btn.textContent = "⏳ A processar venda...";
    btn.disabled = true;

    try {
        // 1. Cria o registo da venda
        const { data: venda, error: errVenda } = await clienteSupabase
            .from('pdv_vendas')
            .insert([{
                turno_id: turnoAtivoGlobal.id,
                total_bruto: totalGeral,
                total_desconto: 0,
                total_liquido: totalGeral,
                status: 'concluida',
                forma_pagamento: formaPgto
            }])
            .select('id, numero_pedido')
            .single();

        if (errVenda) throw errVenda;

        // 2. Insere os itens da venda e dá baixa no stock
        for (const item of carrinhoPdv) {
            await clienteSupabase.from('pdv_itens_venda').insert([{
                venda_id: venda.id,
                produto_id: item.id,
                quantidade: item.quantidade,
                preco_unitario: item.preco_venda,
                subtotal: item.subtotal
            }]);

            // Atualiza stock do produto
            const novoEstoque = item.estoque_atual - item.quantidade;
            await clienteSupabase.from('produtos').update({ estoque_atual: novoEstoque }).eq('id', item.id);
        }

        // 3. Regista o pagamento no financeiro de vendas
        const troco = valorRecebido - totalGeral;
        await clienteSupabase.from('fin_pagamentos_venda').insert([{
            venda_id: venda.id,
            turno_id: turnoAtivoGlobal.id,
            metodo: formaPgto,
            valor_pago: valorRecebido,
            troco: troco
        }]);

        // 4. Cria título no Contas a Receber (fin_contas)
        await clienteSupabase.from('fin_contas').insert([{
            tipo_conta: 'entrada',
            status_conta: 'quitada',
            descricao: `Venda PDV #${venda.id.split('-')[0].toUpperCase()}`,
            valor_nominal: totalGeral,
            valor_pago: totalGeral,
            data_emissao: new Date().toISOString().split('T')[0],
            data_vencimento: new Date().toISOString().split('T')[0],
            data_pagamento: new Date().toISOString().split('T')[0],
            numero_parcela: 1,
            total_parcelas: 1
        }]);

        mostrarToast("Venda finalizada com sucesso!", "sucesso");
        imprimirCupomTermico(venda.id, totalGeral, valorRecebido, troco, formaPgto);
        desenharTelaCaixaOperacao();
    } catch (e) {
        tratarErroSupabase(e, "Finalizar Venda");
        btn.textContent = "✅ Finalizar Venda";
        btn.disabled = false;
    }
}
```

```
# Aperte [Enter] (deixe tudo em branco).
```

```
cat ~/.ssh/id_ed25519.pub
```

* Copie todo esse código que apareceu (começando em ssh-ed25519 até o final do seu e-mail) e adicione-o em **Settings > SSH and GPG keys > New SSH key** no seu GitHub.
* exemplo: ssh-ed255...atkeWeHiX0 aristidesbp@gmail.com
* após salvar tem que confirmar por email.

# CRIANDO UM SERVIDOR PYTHON 
# Passo 1: Instalar o Python3
 * INSTALAR o Termux  (CONFIGURAR E ATUALIZAR)
 * Navegue até a pasta onde seus arquivos estão 
 * caso queira encerrar o processo basta apertar Ctr+C;
 * O codigo abaixo inicia um servidor web simples na porta 8080:
   
```
# CODIGO PARA INICIAR O SERVIDOR:
python3 -m http.server 8080
```

## Como Acessar o Site no Navegador
* Abra o brawser do seu aparelho e digitar:
  
```
http://localhost:8080
```

## ⚠️ Observações importantes
* O servidor só funciona enquanto o Termux estiver aberto.
* para trocar de porta utilize
  
```
python3 -m http.server 3000
```

* Abra o brawser do seu aparelho e digitar:
  
```
 http://localhost:3000
``` 

## ✅ Se quiser acessar de outro dispositivo na mesma rede Wi-Fi
* Vai aparecer algo como: inet 192.168.1.105
* No navegador do outro dispositivo, acesse: http://192.168.1.105:8080
