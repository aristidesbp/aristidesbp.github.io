# SQL PARA VERIFICAR SE EXISTE TABELAS, RLS, RPC, FUNCTIONS E TRIGGER 
```
-- [INÍCIO: EXTRATOR_DE_SCHEMA_SUPABASE]
-- Este script consulta os metadados do PostgreSQL para criar um raio-x do seu schema 'public'

WITH tables_info AS (
    -- Busca todas as tabelas públicas e junta as suas colunas em uma lista
    SELECT jsonb_agg(jsonb_build_object(
        'tabela', t.table_name,
        'colunas', (
            SELECT jsonb_agg(c.column_name || ' (' || c.data_type || ')')
            FROM information_schema.columns c
            WHERE c.table_name = t.table_name AND c.table_schema = 'public'
        )
    )) AS dados
    FROM information_schema.tables t
    WHERE t.table_schema = 'public'
),
rls_info AS (
    -- Busca todas as regras de segurança RLS (Row Level Security)
    SELECT COALESCE(jsonb_agg(jsonb_build_object(
        'tabela', tablename,
        'politica', policyname,
        'comando', cmd,
        'roles', roles,
        'condicao', qual
    )), '[]'::jsonb) AS dados
    FROM pg_policies
    WHERE schemaname = 'public'
),
rpc_info AS (
    -- Busca todas as Funções (RPCs) criadas no schema público
    SELECT COALESCE(jsonb_agg(jsonb_build_object(
        'funcao', p.proname,
        'argumentos', pg_get_function_arguments(p.oid),
        'retorno', pg_get_function_result(p.oid)
    )), '[]'::jsonb) AS dados
    FROM pg_proc p
    JOIN pg_namespace n ON p.pronamespace = n.oid
    WHERE n.nspname = 'public'
),
triggers_info AS (
    -- Busca todos os Triggers (gatilhos) atrelados às tabelas
    SELECT COALESCE(jsonb_agg(jsonb_build_object(
        'tabela', event_object_table,
        'trigger', trigger_name,
        'evento', event_manipulation,
        'tempo', action_timing
    )), '[]'::jsonb) AS dados
    FROM information_schema.triggers
    WHERE trigger_schema = 'public'
)
-- Compila todas as informações acima em um único objeto JSON formatado
SELECT jsonb_pretty(jsonb_build_object(
    '1_tabelas', (SELECT dados FROM tables_info),
    '2_rls_politicas', (SELECT dados FROM rls_info),
    '3_funcoes_rpc', (SELECT dados FROM rpc_info),
    '4_gatilhos_triggers', (SELECT dados FROM triggers_info)
)) AS relatorio_completo;

-- [FIM: EXTRATOR_DE_SCHEMA_SUPABASE]


```

# Verificar storage 
```

-- [INÍCIO: EXTRATOR_POLITICAS_STORAGE_CORRIGIDO]
-- Este script busca as políticas de segurança RLS aplicadas aos arquivos (storage.objects)

WITH storage_policies AS (
    -- Passo 1: Separamos e renomeamos as colunas da tabela de políticas do Postgres
    SELECT
        policyname AS nome_da_politica,
        cmd AS acao_permitida, -- Pode ser SELECT, INSERT, UPDATE, DELETE
        roles AS papeis_permitidos, -- Quem pode fazer isso (ex: authenticated, anon)
        qual AS condicao -- Qual a regra lógica para permitir
    FROM pg_policies
    WHERE schemaname = 'storage' AND tablename = 'objects'
)
-- Passo 2: Juntamos tudo em um formato JSON. 
SELECT COALESCE(
    jsonb_pretty(jsonb_agg(jsonb_build_object(
        'politica', nome_da_politica,
        'acao', acao_permitida,
        'roles', papeis_permitidos,
        'condicao', condicao
    ))), 
    '[]'::text
) AS relatorio_seguranca_storage
FROM storage_policies; -- <- Aqui está a correção! Avisamos de onde puxar os dados.

-- [FIM: EXTRATOR_POLITICAS_STORAGE_CORRIGIDO]
``` 

```
-- 1. Auditoria de Produtos
CREATE OR REPLACE FUNCTION public.registrar_auditoria() RETURNS TRIGGER AS $$
BEGIN
    IF (TG_OP = 'INSERT') THEN
        INSERT INTO public.auditoria_produtos (produto_id, acao, dados_novos, usuario_id) VALUES (NEW.id, 'INSERT', row_to_json(NEW)::jsonb, auth.uid()); RETURN NEW;
    ELSIF (TG_OP = 'UPDATE') THEN
        INSERT INTO public.auditoria_produtos (produto_id, acao, dados_antigos, dados_novos, usuario_id) VALUES (NEW.id, 'UPDATE', row_to_json(OLD)::jsonb, row_to_json(NEW)::jsonb, auth.uid()); RETURN NEW;
    ELSIF (TG_OP = 'DELETE') THEN
        INSERT INTO public.auditoria_produtos (produto_id, acao, dados_antigos, usuario_id) VALUES (OLD.id, 'DELETE', row_to_json(OLD)::jsonb, auth.uid()); RETURN OLD;
    END IF;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
CREATE TRIGGER trigger_auditoria_produtos AFTER INSERT OR UPDATE OR DELETE ON public.produtos FOR EACH ROW EXECUTE FUNCTION public.registrar_auditoria();

-- 2. Automação de Estoque
CREATE OR REPLACE FUNCTION public.automacao_estoque_pdv() RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO public.est_movimentacoes (produto_id, venda_id, tipo, quantidade, observacao) VALUES (NEW.produto_id, NEW.venda_id, 'saida_venda', -NEW.quantidade, 'Baixa PDV');
    UPDATE public.produtos SET estoque_atual = estoque_atual - NEW.quantidade, updated_at = now() WHERE id = NEW.produto_id;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;
CREATE TRIGGER on_venda_item_inserido AFTER INSERT ON public.pdv_itens_venda FOR EACH ROW EXECUTE FUNCTION public.automacao_estoque_pdv();

-- 3. Automação de Fidelidade
CREATE OR REPLACE FUNCTION public.automacao_fidelidade() RETURNS TRIGGER AS $$
BEGIN
    IF NEW.status = 'concluida' AND OLD.status != 'concluida' AND NEW.cliente_id IS NOT NULL THEN
        UPDATE public.entidades SET pontos_fidelidade = pontos_fidelidade + FLOOR(NEW.total_liquido) WHERE id = NEW.cliente_id;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;
CREATE TRIGGER on_venda_concluida AFTER UPDATE ON public.pdv_vendas FOR EACH ROW EXECUTE FUNCTION public.automacao_fidelidade();

-- 4. Blindagem Fiscal (Não alterar venda concluída)
CREATE OR REPLACE FUNCTION public.blindagem_fiscal_venda() RETURNS TRIGGER AS $$
BEGIN
    IF OLD.status = 'concluida' AND NEW.status != 'cancelada' THEN
        RAISE EXCEPTION 'FRAUDE BLOQUEADA: Vendas concluídas não podem ser alteradas.';
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;
CREATE TRIGGER on_alteracao_venda BEFORE UPDATE ON public.pdv_vendas FOR EACH ROW EXECUTE FUNCTION public.blindagem_fiscal_venda();
```

```
/**
 * Recebe um ficheiro de imagem, redimensiona-o e converte para WebP de alta eficiência.
 * @param {File} file - O ficheiro de imagem original.
 * @param {number} maxWidthHeight - A medida máxima permitida (Padrão: 800px).
 * @returns {Promise<File>} - Uma Promise que devolve o ficheiro comprimido.
 */
function comprimirParaWebP(file, maxWidthHeight = 800) {
    return new Promise((resolve, reject) => {
        if (!file) {
            reject(new Error("Nenhum ficheiro fornecido para compressão."));
            return;
        }

        // 1. Lemos o ficheiro original
        const reader = new FileReader();
        reader.readAsDataURL(file);
        
        reader.onload = (event) => {
            // 2. Criamos uma imagem virtual para ler as dimensões reais
            const img = new Image();
            img.src = event.target.result;
            
            img.onload = () => {
                // 3. Lógica para recalcular as proporções sem distorcer a foto
                let width = img.width;
                let height = img.height;

                if (width > height) {
                    if (width > maxWidthHeight) {
                        height = Math.round(height * (maxWidthHeight / width));
                        width = maxWidthHeight;
                    }
                } else {
                    if (height > maxWidthHeight) {
                        width = Math.round(width * (maxWidthHeight / height));
                        height = maxWidthHeight;
                    }
                }

                // 4. Utilizamos o Canvas HTML5 para redesenhar a imagem menor
                const canvas = document.createElement('canvas');
                canvas.width = width;
                canvas.height = height;
                const ctx = canvas.getContext('2d');
                ctx.drawImage(img, 0, 0, width, height);

                // 5. Convertemos a tela para o formato final (.webp a 80%)
                canvas.toBlob((blob) => {
                    if (blob) {
                        // Limpamos a extensão antiga e forçamos a extensão .webp
                        const novoNome = file.name.replace(/\.[^/.]+$/, ".webp");
                        const ficheiroComprimido = new File([blob], novoNome, { type: 'image/webp' });
                        resolve(ficheiroComprimido);
                    } else {
                        reject(new Error("Falha técnica ao converter a imagem."));
                    }
                }, 'image/webp', 0.8);
            };
            
            img.onerror = () => reject(new Error("Ficheiro corrompido ou formato não suportado."));
        };
        
        reader.onerror = () => reject(new Error("Ocorreu um erro na leitura do dispositivo."));
    });
}
```

```
async function excluirEntidadeIndividual(id, nome, fotoUrl) {
    if (!confirm(`Atenção: Tem certeza que deseja apagar ${nome} definitivamente? (Isto falhará se existirem produtos associados a este fornecedor).`)) return;
    try {
        // 1. Apaga a linha da base de dados
        const { error } = await clienteSupabase.from('entidades').delete().eq('id', id);
        if (error) throw error;

        // 2. Apaga a foto do Storage (O Conceito reaproveitado do código antigo)
        if (fotoUrl) {
            const nomeArquivo = decodeURIComponent(fotoUrl.split('/').pop().split('?')[0]);
            await clienteSupabase.storage.from('foto_entidades').remove([nomeArquivo]);
            console.log("Ficheiro limpo do storage:", nomeArquivo);
        }

        alert("✅ Entidade e fotografia apagadas com sucesso!");
        carregarListaEntidades();
    } catch (erro) {
        console.error(erro);
        alert("❌ Erro ao apagar. Pode existir uma restrição no sistema (ex: produtos ligados a este fornecedor).");
    }
}

async function excluirEntidadesEmLote() {
    const chks = document.querySelectorAll('.chk-ent-item:not(#ent-chk-todos):checked');
    const ids = Array.from(chks).map(chk => chk.value);
    if (ids.length === 0 || !confirm(`Aviso: Confirma a eliminação definitiva de ${ids.length} registos e as suas fotos?`)) return;

    try {
        // 1. Resgatar as URLs das fotos ANTES de apagar os dados da tabela
        const { data: entidadesParaApagar } = await clienteSupabase
            .from('entidades')
            .select('foto_url')
            .in('id', ids);

        // 2. Apagar da Base de Dados
        const { error } = await clienteSupabase.from('entidades').delete().in('id', ids);
        if (error) throw error;

        // 3. Limpar o Storage em Lote
        if (entidadesParaApagar && entidadesParaApagar.length > 0) {
            const arquivosParaApagar = entidadesParaApagar
                .filter(ent => ent.foto_url)
                .map(ent => decodeURIComponent(ent.foto_url.split('/').pop().split('?')[0]));
            
            if (arquivosParaApagar.length > 0) {
                await clienteSupabase.storage.from('foto_entidades').remove(arquivosParaApagar);
                console.log("Ficheiros em lote limpos do storage.");
            }
        }

        alert("✅ Registos e fotos apagados em lote com sucesso.");
        document.getElementById('ent-chk-todos').checked = false;
        carregarListaEntidades();
    } catch (erro) {
        console.error(erro);
        alert("❌ Falha técnica ao apagar em lote.");
    }
}
```

```
            document.getElementById('ent-nome').value = data.nome_completo || '';
            document.getElementById('ent-tipo-doc').value = data.tipo_documento || '';

        let urlFinalDaFoto = urlFotoAntiga;
        let nomeFicheiroRecemCarregado = null;

        try {
            if (ficheiroAProcessar) {
                const imagemComprimida = await comprimirParaWebP(ficheiroAProcessar);
                const nomeUnicoSeguro = gerarNomeSeguroArquivo(document.getElementById('ent-nome').value);
                
                const { error: uploadError } = await clienteSupabase.storage
                    .from('foto_entidades')
                    .upload(nomeUnicoSeguro, imagemComprimida);

                if (uploadError) throw uploadError;
                nomeFicheiroRecemCarregado = nomeUnicoSeguro;

                const { data: urlData } = clienteSupabase.storage
                    .from('foto_entidades').getPublicUrl(nomeUnicoSeguro);
                urlFinalDaFoto = urlData.publicUrl;
            }

            const payload = {
                tipo: document.getElementById('ent-tipo').value,
                nome_completo: document.getElementById('ent-nome').value,
                tipo_documento: document.getElementById('ent-tipo-doc').value,
                numero_documento: document.getElementById('ent-documento').value,

                email: document.getElementById('ent-email').value,
                endereco_completo: document.getElementById('ent-endereco').value,
                foto_url: urlFinalDaFoto
            };

            if (idEdicao) {
                const { error } = await clienteSupabase.from('entidades').update(payload).eq('id', idEdicao);
                if (error) throw error;
                if (urlFotoAntiga && nomeFicheiroRecemCarregado) {
                    const nomeVelho = decodeURIComponent(urlFotoAntiga.split('/').pop().split('?')[0]);
                    await clienteSupabase.storage.from('foto_entidades').remove([nomeVelho]);
                }
            } else {
                const { error } = await clienteSupabase.from('entidades').insert([payload]);
                if (error) throw error;
            }

            alert("✅ Entidade gravada com sucesso!");
            desenharModuloClientes();

        } catch (erro) {
            console.error(erro);
            if (nomeFicheiroRecemCarregado) {
                await clienteSupabase.storage.from('foto_entidades').remove([nomeFicheiroRecemCarregado]);
            }
            alert("❌ Ocorreu um erro ao gravar a entidade.");
        } finally {
            btnSalvar.textContent = "💾 Salvar Entidade";
            btnSalvar.disabled = false;
        }
    });
```
