```
    p_peso_liquido numeric DEFAULT 0,
    p_dimensoes text DEFAULT NULL::text,

) RETURNS uuid -- A função vai devolver o ID (UUID) do produto acabado de criar

AS $$
DECLARE
    v_novo_id uuid;
    v_forn_uuid uuid := NULL;
BEGIN
    -- Validação: Se o fornecedor foi enviado (não está vazio), convertemos de texto para UUID
    IF p_fornecedor IS NOT NULL AND p_fornecedor <> '' THEN
        v_forn_uuid := p_fornecedor::uuid;
    END IF;

    -- Inserimos os dados na tabela produtos
    INSERT INTO public.produtos (
        nome, 
        preco_venda, 
        foto_url, 
        sku, 
        codigo_barras,
        ncm, 
        cest, 
        cfop_padrao, 
        unidade_comercial, 
        estoque_atual,
        preco_custo, 
        categoria, 
        fornecedor_id
    ) VALUES (
        p_nome, 
        p_preco, 
        p_foto_url, 
        p_codigo_produto, 
        p_ean,
        p_ncm, 
        p_cest, 
        p_cfop, 
        p_unidade_comercial, 
        p_quantidade,
        p_preco_custo, 
        p_categoria, 
        v_forn_uuid
    ) RETURNING id INTO v_novo_id; -- Captura o ID gerado automaticamente

    -- Devolve o ID para o teu JavaScript (produtos.js)
    RETURN v_novo_id;
END;

    p_preco_custo numeric DEFAULT 0,
    p_estoque_minimo numeric DEFAULT 0,
    p_origem text DEFAULT NULL::text,
    p_local_estoque text DEFAULT NULL::text,
    p_local_loja text DEFAULT NULL::text,
    p_tipo_embalagem text DEFAULT NULL::text,
    p_categoria text DEFAULT NULL::text,
    p_peso_liquido numeric DEFAULT 0,
    p_dimensoes text DEFAULT NULL::text,
    p_data_compra date DEFAULT NULL::date,
    p_data_vencimento date DEFAULT NULL::date
) RETURNS void -- Esta função não precisa devolver nada (void), apenas atualiza
LANGUAGE plpgsql

AS $$
DECLARE
    v_forn_uuid uuid := NULL;
BEGIN
    -- Validação do fornecedor
    IF p_fornecedor IS NOT NULL AND p_fornecedor <> '' THEN
        v_forn_uuid := p_fornecedor::uuid;
    END IF;

    -- Atualizamos o registo que tenha o ID correspondente
    UPDATE public.produtos
    SET
        nome = p_nome,
        preco_venda = p_preco,
        foto_url = COALESCE(p_foto_url, foto_url), -- Mantém a foto antiga se nenhuma nova for enviada
        sku = p_codigo_produto,
        codigo_barras = p_ean,
        ncm = p_ncm,
        cest = p_cest,
        cfop_padrao = p_cfop,
        unidade_comercial = p_unidade_comercial,
        estoque_atual = p_quantidade,
        preco_custo = p_preco_custo,
        categoria = p_categoria,
        fornecedor_id = v_forn_uuid,
        updated_at = now() -- Atualiza a data de modificação
    WHERE id = p_id;
END;
$$;
``` 

```
-- [INÍCIO: EXTRATOR_DE_SCHEMA_SUPABASE]
-- Este script consulta os metadados do PostgreSQL para criar um raio-x do seu schema 'public'

WITH tables_info AS (
    -- Busca todas as tabelas públicas e junta as suas colunas em uma lista
    SELECT jsonb_agg(jsonb_build_object(
        'tabela', t.table_name,
        'colunas', (
            SELECT jsonb_agg(c.column_name || ' (' || c.data_type || ')')
            FROM information_schema.columns c
            WHERE c.table_name = t.table_name AND c.table_schema = 'public'
        )
    )) AS dados
    FROM information_schema.tables t
    WHERE t.table_schema = 'public'
),
rls_info AS (
    -- Busca todas as regras de segurança RLS (Row Level Security)
    SELECT COALESCE(jsonb_agg(jsonb_build_object(
        'tabela', tablename,
        'politica', policyname,
        'comando', cmd,
        'roles', roles,
        'condicao', qual
    )), '[]'::jsonb) AS dados
    FROM pg_policies
    WHERE schemaname = 'public'
),
rpc_info AS (
    -- Busca todas as Funções (RPCs) criadas no schema público
    SELECT COALESCE(jsonb_agg(jsonb_build_object(
        'funcao', p.proname,
        'argumentos', pg_get_function_arguments(p.oid),
        'retorno', pg_get_function_result(p.oid)
    )), '[]'::jsonb) AS dados
    FROM pg_proc p
    JOIN pg_namespace n ON p.pronamespace = n.oid
    WHERE n.nspname = 'public'
),
triggers_info AS (
    -- Busca todos os Triggers (gatilhos) atrelados às tabelas
    SELECT COALESCE(jsonb_agg(jsonb_build_object(
        'tabela', event_object_table,
        'trigger', trigger_name,
        'evento', event_manipulation,
        'tempo', action_timing
    )), '[]'::jsonb) AS dados
    FROM information_schema.triggers
    WHERE trigger_schema = 'public'
)
-- Compila todas as informações acima em um único objeto JSON formatado
SELECT jsonb_pretty(jsonb_build_object(
    '1_tabelas', (SELECT dados FROM tables_info),
    '2_rls_politicas', (SELECT dados FROM rls_info),
    '3_funcoes_rpc', (SELECT dados FROM rpc_info),
    '4_gatilhos_triggers', (SELECT dados FROM triggers_info)
)) AS relatorio_completo;
```

```
function comprimirParaWebP(file, maxWidthHeight = 800) {
    return new Promise((resolve, reject) => {
        if (!file) return reject(new Error("Nenhum ficheiro fornecido."));
        const reader = new FileReader();
        reader.readAsDataURL(file);
        reader.onload = (event) => {
            const img = new Image();
            img.src = event.target.result;
            img.onload = () => {
                let width = img.width;
                let height = img.height;
                if (width > height) {
                    if (width > maxWidthHeight) { height = Math.round(height * (maxWidthHeight / width)); width = maxWidthHeight; }
                } else {
                    if (height > maxWidthHeight) { width = Math.round(width * (maxWidthHeight / height)); height = maxWidthHeight; }
                }
                const canvas = document.createElement('canvas');
                canvas.width = width;
                canvas.height = height;
                const ctx = canvas.getContext('2d');
                ctx.drawImage(img, 0, 0, width, height);
                canvas.toBlob((blob) => {
                    if (blob) {
                        const novoNome = file.name.replace(/\.[^/.]+$/, ".webp");
                        resolve(new File([blob], novoNome, { type: 'image/webp' }));
                    } else {
                        reject(new Error("Falha ao converter imagem para WebP."));
                    }
                }, 'image/webp', 0.8);
            };
            img.onerror = () => reject(new Error("Formato de imagem inválido."));
        };
        reader.onerror = () => reject(new Error("Erro ao ler ficheiro."));
    });
}

function gerarNomeSeguroArquivo(prefixo) {
    let nomeLimpo = (prefixo || 'arq').normalize('NFD').replace(/[\u0300-\u036f]/g, "").toLowerCase();
    nomeLimpo = nomeLimpo.replace(/[^a-z0-9]/g, '-').replace(/-+/g, '-');
    return `${nomeLimpo}-${Date.now()}.webp`;
}
```

# CRIANDO UM SERVIDOR PYTHON 
# Passo 1: Instalar o Python3
 * INSTALAR o Termux  (CONFIGURAR E ATUALIZAR)
 * Navegue até a pasta onde seus arquivos estão 
 * caso queira encerrar o processo basta apertar Ctr+C;
 * O codigo abaixo inicia um servidor web simples na porta 8080:
   
```
# CODIGO PARA INICIAR O SERVIDOR:
python3 -m http.server 8080
```

## Como Acessar o Site no Navegador
* Abra o brawser do seu aparelho e digitar:
  
```
http://localhost:8080
```

## ⚠️ Observações importantes
* O servidor só funciona enquanto o Termux estiver aberto.
* para trocar de porta utilize
  
```
python3 -m http.server 3000
```

* Abra o brawser do seu aparelho e digitar:
  
```
 http://localhost:3000
``` 

## ✅ Se quiser acessar de outro dispositivo na mesma rede Wi-Fi
* Vai aparecer algo como: inet 192.168.1.105
* No navegador do outro dispositivo, acesse: http://192.168.1.105:8080
  
```
# Descubra o IP do celular no Termux:
```
