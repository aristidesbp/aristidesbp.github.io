-- BACKUP DEFINITIVO DO PROJETO - CRUD DE EQUIPAMENTOS TI
-- =======================================================================================


-- =======================================================================================
-- 1. CRIAÇÃO DA TABELA PRINCIPAL
-- =======================================================================================
CREATE TABLE public.equipamentos_ti (
    id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
    nome text NOT NULL,
    preco numeric NOT NULL,
    especificacoes jsonb DEFAULT '{}'::jsonb,
    foto_url text DEFAULT NULL,
    esta_ativo boolean DEFAULT true,
    criado_em timestamp with time zone DEFAULT now()
);


-- =======================================================================================
-- 2. SEGURANÇA A NÍVEL DE LINHA (RLS) DA TABELA
-- =======================================================================================
-- Ativa a barreira de proteção na tabela
ALTER TABLE public.equipamentos_ti ENABLE ROW LEVEL SECURITY;

-- Cria a regra que permite ao Frontend LER apenas os itens que estão ativos
CREATE POLICY "Permitir leitura publica de equipamentos ativos" 
ON public.equipamentos_ti
FOR SELECT TO public
USING (esta_ativo = true);


-- =======================================================================================
-- 3. POLÍTICAS DE SEGURANÇA DO STORAGE (PASTA: equipamentos_fotos)
-- =======================================================================================
-- 3.1. Permite Ler/Buscar o arquivo (Necessário para a exclusão física funcionar)
DROP POLICY IF EXISTS "Permitir leitura de imagens para exclusao" ON storage.objects;
CREATE POLICY "Permitir leitura de imagens para exclusao" 
ON storage.objects
FOR SELECT TO public
USING (bucket_id = 'equipamentos_fotos');

-- 3.2. Permite Inserir novos arquivos (Upload)
DROP POLICY IF EXISTS "Permitir upload de imagens publicas" ON storage.objects;
CREATE POLICY "Permitir upload de imagens publicas" 
ON storage.objects
FOR INSERT TO public
WITH CHECK (bucket_id = 'equipamentos_fotos');

-- 3.3. Permite Excluir arquivos fisicamente (Delete)
DROP POLICY IF EXISTS "Permitir exclusao de imagens publicas" ON storage.objects;
CREATE POLICY "Permitir exclusao de imagens publicas" 
ON storage.objects
FOR DELETE TO public
USING (bucket_id = 'equipamentos_fotos');


-- =======================================================================================
-- 4. FUNÇÕES SEGURAS (RPCs) - REGRAS DE NEGÓCIO NO BACKEND
-- =======================================================================================

-- 4.1. CREATE (Adicionar Equipamento)
CREATE OR REPLACE FUNCTION public.adicionar_equipamento_seguro(
    p_nome text,
    p_preco numeric,
    p_especificacoes jsonb,
    p_foto_url text DEFAULT NULL
)
RETURNS uuid
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
    v_novo_id uuid;
BEGIN
    IF p_nome IS NULL OR length(trim(p_nome)) < 3 THEN
        RAISE EXCEPTION 'Erro de Segurança: O nome do equipamento deve ter pelo menos 3 caracteres.';
    END IF;

    IF p_preco <= 0 THEN
        RAISE EXCEPTION 'Erro de Segurança: O preço deve ser maior que zero.';
    END IF;

    INSERT INTO public.equipamentos_ti (nome, preco, especificacoes, foto_url, esta_ativo)
    VALUES (trim(p_nome), p_preco, p_especificacoes, p_foto_url, true)
    RETURNING id INTO v_novo_id;

    RETURN v_novo_id;
END;
$$;


-- 4.2. UPDATE (Atualizar Equipamento)
CREATE OR REPLACE FUNCTION public.atualizar_equipamento_seguro(
    p_id uuid,
    p_novo_nome text,
    p_novo_preco numeric,
    p_foto_url text DEFAULT NULL
)
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
BEGIN
    IF p_novo_nome IS NULL OR length(trim(p_novo_nome)) < 3 THEN
        RAISE EXCEPTION 'Erro de Segurança: O nome do equipamento deve ter pelo menos 3 caracteres.';
    END IF;

    IF p_novo_preco <= 0 THEN
        RAISE EXCEPTION 'Erro de Segurança: O preço deve ser maior que zero.';
    END IF;

    UPDATE public.equipamentos_ti
    SET 
        nome = trim(p_novo_nome),
        preco = p_novo_preco,
        foto_url = p_foto_url
    WHERE id = p_id AND esta_ativo = true;
END;
$$;


-- 4.3. SOFT DELETE (Ocultar Equipamento individualmente)
CREATE OR REPLACE FUNCTION public.desativar_equipamento_seguro(
    p_id uuid
)
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
BEGIN
    UPDATE public.equipamentos_ti
    SET esta_ativo = false
    WHERE id = p_id;
END;
$$;


-- 4.4. HARD DELETE EM LOTE (Esvaziar Lixeira e retornar URLs das fotos)
CREATE OR REPLACE FUNCTION public.limpar_lixeira_seguro()
RETURNS TABLE(url_da_foto text)
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
BEGIN
    RETURN QUERY
    DELETE FROM public.equipamentos_ti
    WHERE esta_ativo = false
    RETURNING foto_url;
END;
$$;



-- Cria o Bucket (Pasta) de fotos via SQL e o define como público
INSERT INTO storage.buckets (id, name, public)
VALUES ('equipamentos_fotos', 'equipamentos_fotos', true)
ON CONFLICT (id) DO NOTHING;

-- =======================================================================================
-- BLINDAGEM DEFINITIVA: RLS RESTRITO APENAS A USUÁRIOS LOGADOS (AUTHENTICATED)
-- =======================================================================================

-- 1. REMOVER POLÍTICAS ANTIGAS PÚBLICAS
DROP POLICY IF EXISTS "Permitir leitura publica de equipamentos ativos" ON public.equipamentos_ti;
DROP POLICY IF EXISTS "Permitir leitura de imagens para exclusao" ON storage.objects;
DROP POLICY IF EXISTS "Permitir upload de imagens publicas" ON storage.objects;
DROP POLICY IF EXISTS "Permitir exclusao de imagens publicas" ON storage.objects;

-- 2. NOVAS POLÍTICAS PARA A TABELA (Somente usuários logados)
CREATE POLICY "Apenas autenticados leem equipamentos" 
ON public.equipamentos_ti FOR SELECT TO authenticated 
USING (esta_ativo = true);

-- As funções RPC rodam com SECURITY DEFINER, mas garantimos a segurança geral:
CREATE POLICY "Apenas autenticados modificam tabela" 
ON public.equipamentos_ti FOR ALL TO authenticated 
USING (true) WITH CHECK (true);

-- 3. NOVAS POLÍTICAS PARA O STORAGE (Somente usuários logados gerenciam arquivos)
CREATE POLICY "Apenas autenticados leem fotos" 
ON storage.objects FOR SELECT TO authenticated 
USING (bucket_id = 'equipamentos_fotos');

CREATE POLICY "Apenas autenticados sobem fotos" 
ON storage.objects FOR INSERT TO authenticated 
WITH CHECK (bucket_id = 'equipamentos_fotos');

CREATE POLICY "Apenas autenticados apagam fotos" 
ON storage.objects FOR DELETE TO authenticated 
USING (bucket_id = 'equipamentos_fotos');





-- =======================================================================================
-- CORREÇÃO DA RLS: SEPARAR LEITURA DAS DEMAIS OPERAÇÕES
-- =======================================================================================

-- 1. Removemos as políticas anteriores que entraram em conflito
DROP POLICY IF EXISTS "Apenas autenticados leem equipamentos" ON public.equipamentos_ti;
DROP POLICY IF EXISTS "Apenas autenticados modificam tabela" ON public.equipamentos_ti;

-- 2. Regra restrita EXCLUSIVAMENTE para a Leitura (SELECT)
-- Garante que usuários logados só enxerguem o que está ativo (esta_ativo = true)
CREATE POLICY "Apenas autenticados leem equipamentos ativos" 
ON public.equipamentos_ti 
FOR SELECT 
TO authenticated 
USING (esta_ativo = true);

-- 3. Regras separadas para Inserção, Atualização e Exclusão
CREATE POLICY "Apenas autenticados inserem equipamentos" 
ON public.equipamentos_ti 
FOR INSERT 
TO authenticated 
WITH CHECK (true);

CREATE POLICY "Apenas autenticados atualizam equipamentos" 
ON public.equipamentos_ti 
FOR UPDATE 
TO authenticated 
USING (true) 
WITH CHECK (true);

CREATE POLICY "Apenas autenticados deletam equipamentos" 
ON public.equipamentos_ti 
FOR DELETE 
TO authenticated 
USING (true);
